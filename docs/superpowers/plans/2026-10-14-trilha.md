**Goal:** cada Short sai com uma cama instrumental da biblioteca, escolhida pelo tom do trecho
(LLM) ou à mão no editor, 18 LU abaixo da voz, tocando na prévia e no MP4.

**Architecture:** contas puras em `render/trilha.py` (ganho, clipes, HTML); a biblioteca e a

`<audio>` no `index.html` do HyperFrames (placeholder `{{TRILHA}}` nos 7 templates). A escolha é
`Edicoes.trilha`, editada por `POST …/editar`.

**Tech Stack:** Python 3.13, FastAPI, pydantic, ffmpeg (`ebur128`), HyperFrames 0.8.104,
`claude -p` (via `cortes.llm`), JS puro no painel.

```python
"""Trilha de fundo do Short: contas puras de nível, clipes e HTML (spec trilha §5).

Regras de `pesquisa/04-trilha.md`: a cama fica ALVO_LU abaixo da voz, cada uma medida sozinha em
LUFS integrado (R1); entra com fade curto (R4) e sai num compasso inteiro que termina no fim do
vídeo (R5). Volume fixo, sem sidechain: as pausas já foram cortadas (R2). `data-volume` do
HyperFrames é ganho linear (`@hyperframes/core` 0.8.104, dist/audioGain.js).
"""
import math
import re
from dataclasses import dataclass

from cortes.render.processo import CAUDA, ErroRender

ALVO_LU = 18.0          # R1: voz 18 LU acima da cama
FADE_IN_S = 0.3         # R4
SOBREPOSICAO_S = 2.0    # crossfade entre repetições da cama (corte mais longo que ela)
TRACK = 11              # a voz (cam-audio) é a 10; repetições alternam 11/12 para se sobrepor
ARQUIVO_ASSET = "trilha.m4a"
_INTEGRADA = re.compile(r"I:\s+(-?[\d.]+) LUFS")
_TEMPO = re.compile(r"\bt:\s*([\d.]+)")


@dataclass(frozen=True)
class Medida:
    lufs: float
    duracao: float


@dataclass(frozen=True)
class Clipe:
    inicio: float
    duracao: float
    media_inicio: float
    fade_in: float
    fade_out: float


def ler_ebur128(log: str) -> Medida:
    """O último `I: … LUFS` é o resumo (as linhas por quadro também têm `I:`); o último `t:` é a
    duração medida."""
    integradas, tempos = _INTEGRADA.findall(log), _TEMPO.findall(log)
    if not integradas or not tempos:
        raise ErroRender(f"ebur128 sem loudness integrada: {log[-CAUDA:]}")
    return Medida(float(integradas[-1]), float(tempos[-1]))


def ganho(lufs_voz: float, lufs_cama: float, alvo_lu: float = ALVO_LU) -> float:
    """`data-volume` linear para a cama ficar `alvo_lu` abaixo da voz; nunca acima de 1.0 (não
    amplifica a cama)."""
    return min(1.0, 10 ** ((lufs_voz - alvo_lu - lufs_cama) / 20))


def compasso(bpm: float) -> float:
    return 4 * 60 / bpm


def clipes(dur: float, duracao_cama: float, bpm: float) -> list[Clipe]:
    if dur <= 0 or duracao_cama <= 2 * SOBREPOSICAO_S:
        raise ErroRender(f"trilha: duração inválida (corte {dur}s, cama {duracao_cama}s)")
    if dur <= duracao_cama:
        return [_clipe_unico(dur, duracao_cama, compasso(bpm))]
    return _encadeados(dur, duracao_cama, compasso(bpm))


def _clipe_unico(dur: float, duracao_cama: float, barra: float) -> Clipe:
    """Começa no ponto da cama que faz o vídeo terminar numa barra (R5); se essa barra passa do
    fim da cama, começa do zero, sem alinhar."""
    fim = math.ceil(dur / barra - 1e-9) * barra
    media = fim - dur if fim <= duracao_cama + 1e-9 else 0.0
    return Clipe(0.0, round(dur, 3), round(media, 3), FADE_IN_S, round(min(barra, dur / 2), 3))


def _encadeados(dur: float, duracao_cama: float, barra: float) -> list[Clipe]:
    """Repetições da cama a cada `duracao_cama - SOBREPOSICAO_S`, com crossfade de
    SOBREPOSICAO_S. A última sobra é sempre maior que SOBREPOSICAO_S (a anterior não coube)."""
    passo, res, t = duracao_cama - SOBREPOSICAO_S, [], 0.0
    while True:
        resto = dur - t
        ultimo = resto <= duracao_cama
        res.append(Clipe(round(t, 3), round(min(resto, duracao_cama), 3), 0.0,
                         FADE_IN_S if t == 0 else SOBREPOSICAO_S,
                         round(min(barra, resto / 2), 3) if ultimo else SOBREPOSICAO_S))
        if ultimo:
            return res
        t += passo


def html_trilha(cs: list[Clipe], volume: float) -> str:
    return "\n".join(
        f'      <audio id="trilha-{k}" src="assets/{ARQUIVO_ASSET}" data-start="{c.inicio}" '
        f'data-duration="{c.duracao}" data-media-start="{c.media_inicio}" '
        f'data-track-index="{TRACK + k % 2}" data-volume="{volume:.6g}" data-fade-in="{c.fade_in}" '
        f'data-fade-out="{c.fade_out}" data-audio-group="trilha"></audio>'
        for k, c in enumerate(cs))
```

```python
"""Biblioteca de camas da trilha (spec trilha §3): `biblioteca/trilhas/plano.json` + `final/<id>.m4a`.

Cada fase do plano tem `id`, `mood` (um de MOODS) e `bpm`; todo mood precisa de pelo menos uma
cama. A loudness de cada cama é medida uma vez e guardada em `medidas.json`, chaveada pelo tamanho
e mtime do arquivo (arquivo trocado → mede de novo).
"""
from collections.abc import Callable
from dataclasses import asdict, dataclass
from pathlib import Path
from typing import Any

from cortes.episodio import ler_json, salvar_json
from cortes.render.processo import CAUDA, ErroRender, Rodar, rodar_processo
from cortes.render.trilha import Medida, ler_ebur128

BIBLIOTECA = Path("biblioteca") / "trilhas"
MOODS = ("neutro", "tensao", "leve")
MEDIDAS = "medidas.json"
MedirLufs = Callable[[Path], Medida]


@dataclass(frozen=True)
class Cama:
    id: str
    mood: str
    bpm: float
    arquivo: Path


def _cama(pasta: Path, f: Any) -> Cama:
    return Cama(str(f["id"]), str(f["mood"]), float(f["bpm"]), pasta / "final" / f"{f['id']}.m4a")


def ler_plano(pasta: Path) -> list[Cama]:
    arq = pasta / "plano.json"
    if not arq.exists():
        raise ErroRender(f"{arq}: plano.json não existe (a biblioteca de trilhas é obrigatória)")
    try:
        camas = [_cama(pasta, f) for f in ler_json(arq)["fases"]]
    except (KeyError, TypeError, ValueError) as ex:
        raise ErroRender(f"{arq}: plano inválido ({ex})") from ex
    _validar(arq, camas)
    return camas


def _validar(arq: Path, camas: list[Cama]) -> None:
    ids = [c.id for c in camas]
    if len(set(ids)) != len(ids):
        raise ErroRender(f"{arq}: id de cama repetido em {ids}")
    for c in camas:
        if c.mood not in MOODS:
            raise ErroRender(f"{arq}: cama {c.id} com mood '{c.mood}'; use um de {MOODS}")
        if not c.arquivo.exists():
            raise ErroRender(f"{arq}: cama {c.id} sem arquivo ({c.arquivo} não existe)")
    if faltam := [m for m in MOODS if m not in {c.mood for c in camas}]:
        raise ErroRender(f"{arq}: nenhuma cama para o mood {', '.join(faltam)}")


def cama_do_mood(camas: list[Cama], mood: str) -> Cama:
    """A primeira cama do mood (ler_plano garante que existe)."""
    return next(c for c in camas if c.mood == mood)


def cama_por_id(camas: list[Cama], id: str) -> Cama:
    for c in camas:
        if c.id == id:
            return c
    raise ErroRender(f"cama '{id}' não existe na biblioteca; use uma de {[c.id for c in camas]}")


def medir_lufs(arq: Path, rodar: Rodar) -> Medida:
    args = ["ffmpeg", "-hide_banner", "-nostats", "-i", str(arq), "-af", "ebur128", "-f", "null", "-"]
    codigo, log = rodar(args, None)
    if codigo != 0:
        raise ErroRender(f"ffmpeg falhou ao medir {arq}: {log[-CAUDA:]}")
    return ler_ebur128(log)


def medir_lufs_real(arq: Path) -> Medida:
    return medir_lufs(arq, rodar_processo)


def medida_da_cama(cama: Cama, pasta: Path, medir: MedirLufs) -> Medida:
    arq = pasta / MEDIDAS
    cache: dict[str, Any] = ler_json(arq) if arq.exists() else {}
```

```python
Uma chamada `claude -p` por texto de corte, com cache em `cortes/<n>/trilha.json`: renderizar de
novo sem mudar o texto não chama o LLM. Na dúvida, "neutro": música tensa por baixo de fala neutra
muda o julgamento sobre quem fala (Boltz 2001, R8).
"""
from dataclasses import asdict, dataclass
from pathlib import Path
from typing import Any

from cortes.detector import ChamarLLM
from cortes.episodio import ler_json, salvar_json
from cortes.llm import ErroLLM
from cortes.render.processo import ErroRender
from cortes.render.trilha_biblioteca import MOODS

ARQUIVO = "trilha.json"
MODELO = "sonnet"
SISTEMA = """Você classifica o tom de um corte de podcast em português brasileiro para escolher a
música de fundo. Moods:
- "tensao": o trecho causa espanto, medo, raiva ou alerta (perda de dinheiro, crise, golpe, risco).
- "leve": o trecho é engraçado ou descontraído; há piada ou riso.
- "neutro": explicação, opinião ou conversa sem carga forte.
Na dúvida, "neutro": música tensa por baixo de fala neutra faz quem fala parecer suspeito.
O motivo é curto (até 120 caracteres) e cita o que no trecho decidiu."""
SCHEMA: dict[str, Any] = {
    "type": "object",
    "additionalProperties": False,
    "required": ["mood", "motivo"],
    "properties": {
        "mood": {"type": "string", "enum": list(MOODS)},
        "motivo": {"type": "string", "minLength": 1, "maxLength": 200},
    },
}


@dataclass(frozen=True)
class Mood:
    mood: str
    motivo: str


def validar(dados: dict[str, Any]) -> Mood:
    mood, motivo = dados.get("mood"), dados.get("motivo")
    if mood not in MOODS:
        raise ErroRender(f"trilha: mood '{mood}' fora de {MOODS}")
    if not isinstance(motivo, str) or not motivo.strip():
        raise ErroRender(f"trilha: motivo vazio ou ausente ({motivo!r})")
    return Mood(str(mood), motivo.strip())


def ler_mood(d_corte: Path) -> Mood | None:
    arq = d_corte / ARQUIVO
    return validar(ler_json(arq)) if arq.exists() else None


def preparar_mood(d_corte: Path, texto: str, chamar: ChamarLLM) -> Mood:
    arq = d_corte / ARQUIVO
    chave = {"texto": texto}
    if arq.exists():
        lido = ler_json(arq)
        if lido.get("chave") == chave:
            return validar(lido)
    try:
        dados = chamar(f"Fala do corte:\n{texto}", SCHEMA, SISTEMA, MODELO)
    except ErroLLM as e:
        raise ErroRender(f"trilha: {e}") from e
    m = validar(dados)
    d_corte.mkdir(parents=True, exist_ok=True)
    salvar_json(arq, {**asdict(m), "chave": chave})
    return m
```

```python
"""Etapa da trilha no `montar_projeto` (spec trilha §6): resolve a cama (escolha do editor ou mood
do LLM), mede a voz do trecho, copia a cama para `assets/trilha.m4a` e devolve o HTML dos
`<audio>` e o registro do que foi usado."""
import shutil
from dataclasses import dataclass
from pathlib import Path

from cortes.detector import ChamarLLM
from cortes.render.corte import Corte
from cortes.render.edicao import SEM_TRILHA
from cortes.render.trilha import ARQUIVO_ASSET, clipes, ganho, html_trilha
from cortes.render.trilha_biblioteca import (
    Cama,
    MedirLufs,
    cama_do_mood,
    cama_por_id,
    ler_plano,
    medida_da_cama,
)
from cortes.render.trilha_mood import preparar_mood


@dataclass(frozen=True)
class TrilhaUsada:
    cama: str | None
    origem: str            # "auto" (LLM) ou "manual" (editor)
    motivo: str | None     # o do LLM, quando automática
    volume: float | None   # data-volume da cama


def _escolher(d: Path, c: Corte, texto: str, camas: list[Cama],
              chamar: ChamarLLM) -> tuple[Cama, str, str | None]:
    if c.edicoes.trilha is None:
        m = preparar_mood(d, texto, chamar)
        return cama_do_mood(camas, m.mood), "auto", m.motivo
    return cama_por_id(camas, c.edicoes.trilha), "manual", None


def preparar_trilha(d: Path, c: Corte, texto: str, dur: float, biblioteca: Path, chamar: ChamarLLM,
                    medir: MedirLufs) -> tuple[str, TrilhaUsada]:
    assets = d / "projeto" / "assets"
    if c.edicoes.trilha == SEM_TRILHA:
        (assets / ARQUIVO_ASSET).unlink(missing_ok=True)
        return "", TrilhaUsada(None, "manual", None, None)
    cama, origem, motivo = _escolher(d, c, texto, ler_plano(biblioteca), chamar)
    mc = medida_da_cama(cama, biblioteca, medir)
    volume = ganho(medir(assets / "trecho.mp4").lufs, mc.lufs)
    shutil.copyfile(cama.arquivo, assets / ARQUIVO_ASSET)
    html = html_trilha(clipes(dur, mc.duracao, cama.bpm), volume)
    return html, TrilhaUsada(cama.id, origem, motivo, round(volume, 6))
```

```python
    k = c.versao + 1
    renderizar(d / "projeto", d / f"short-v{k}.mp4", dep.npx, dep.rodar)
```
